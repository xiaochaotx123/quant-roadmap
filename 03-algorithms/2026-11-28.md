## 14. 最长公共前缀  

```python
# C++

class Solution {
public:
    string longestCommonPrefix(vector<string>& strs) {

        if(!strs.size()) return "";

        int max_len = 0;
        for(int len = 1; len <= strs[0].size(); len ++){
            bool ok = true;
            for(int j = 0; j < (int)strs.size(); j ++){

                if(strs[j][len - 1] != strs[0][len - 1]){
                    ok = false;;
                    break;
                }
            }
            if(ok) max_len = len;
            else break;
        }

        string ans = "";
        
        for(int i = 0; i < max_len; i ++) ans += strs[0][i];

        return ans;
    }
};
```

```python
# Python

class Solution:
    def longestCommonPrefix(self, strs: list[str]) -> str:
        if len(strs) == 0:
            return ""
        
        max_len = int(0)
        lenth = int(0)

        while True:
            ok = True
            for i in range(0, len(strs), 1):
                if lenth >= len(strs[i]) or strs[0][lenth] != strs[i][lenth]:
                    ok = False
                    break
            if ok:
                lenth += 1
            else:
                break
        
        s = ""
        for i in range(lenth):
            s += strs[0][i]

        return s
```

## 26. 删除有序数组中的重复项

```python
# C++

class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int lpos = 1;
        for(int rpos = 0; rpos < (int)nums.size(); rpos ++){
            if(nums[lpos - 1] != nums[rpos]){
                nums[lpos] = nums[rpos];
                lpos ++;
            }
        }
        return lpos;
    }
};
```

```python
# Python

class Solution:
    def removeDuplicates(self, nums: list[int]) -> int:

        if not nums:
            return 0

        lpos = int(1)
        for rpos in range(len(nums)):
            if nums[lpos - 1] != nums[rpos]:
                nums[lpos] = nums[rpos]
                lpos += 1

        return lpos
```


## 27. 移除元素
```python
# C++

class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int lpos = 0;
        for(int rpos = 0; rpos < (int)nums.size(); rpos ++){
            if(nums[rpos] != val) nums[lpos ++] = nums[rpos];
        }
        return lpos;
    }
};
```

```python
# Python

class Solution:
    def removeElement(self, nums: list[int], val: int) -> int:
        lpos = int(0)
        for num in nums:
            if num != val:
                nums[lpos] = num
                lpos += 1

        return lpos
```

## 35. 搜索插入位置
```python
# C ++

class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int l = 0, r = (int)nums.size() - 1, fpos = -1;
        while(l <= r){
            int mid = (l + r) / 2;
            if(nums[mid] <= target){
                fpos = mid;
                l = mid + 1;
            }
            else r = mid - 1;
        }
        if(fpos >=0 && nums[fpos] == target) return fpos;
        else return fpos + 1;
    }
};
```
