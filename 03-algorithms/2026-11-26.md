#### 3360. 移除石头游戏
```python
# C++
class Solution {
public:
    bool canAliceWin(int n) {
        int state = 1;
        int num = 10;

        while(n >= num){
            n -= num;
            num --;
            state ^= 1;
        }

        if(state) return false;
        else return true;
    }
};
```
```python
# Python
class Solution:
    def canAliceWin(self, n: int) -> bool:
        state = int(1)
        num = int(10)
        while n >= num:
            state ^= 1
            n -= num
            num -= 1

        if not state:
            return True
        else:
            return False
```

#### 1. 两数之和
```python
# C++

class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        map<int, int> ma;
        for(int i = 0; i < (int)nums.size(); i ++){
            if(ma.count(target - nums[i])){
                return {ma[target - nums[i]], i};
            }
            ma[nums[i]] = i;
        }
        return {0, 0};
    }
};


# Python
class Solution:
    def twoSum(self, nums: list[int], target: int) -> list[int]:
        hashmap = {}
        for i, num in enumerate(nums):
            if (target - num) in hashmap:
                return [hashmap[target - num], i]
            hashmap[num] = i
        
        return []
```
#### enumeratr()
`enumerate()` 函数是 Python 中一个非常常用且好用的**内置函数**  

简单来说，它的作用就是遍历一个列表（或其他可迭代对象）时，同时帮你获取元素的 "索引（下标）" 和 "值"  

每次循环时，它会吐出包含两个元素的元组，第一个变量是元素的索引，第二个变量时元素的值      

`enumeratr()` 还可以接受第二个参数 `start`，用于指定索引的起始值，默认是 0，比如：
```python
fruits = ['苹果', '香蕉', '橘子']

for i, fruit in enumerate(fruits, start=1):
    print(f"第 {i} 个水果是: {fruit}")
```
