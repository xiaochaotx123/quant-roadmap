#### 3360. 移除石头游戏
```python
# C++
class Solution {
public:
    bool canAliceWin(int n) {
        int state = 1;
        int num = 10;

        while(n >= num){
            n -= num;
            num --;
            state ^= 1;
        }

        if(state) return false;
        else return true;
    }
};
```
```python
# Python
class Solution:
    def canAliceWin(self, n: int) -> bool:
        state = int(1)
        num = int(10)
        while n >= num:
            state ^= 1
            n -= num
            num -= 1

        if not state:
            return True
        else:
            return False
```

#### 1. 两数之和
```python
# C++

class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        map<int, int> ma;
        for(int i = 0; i < (int)nums.size(); i ++){
            if(ma.count(target - nums[i])){
                return {ma[target - nums[i]], i};
            }
            ma[nums[i]] = i;
        }
        return {0, 0};
    }
};
```
